-- ============================================================ -- SALESPRO - AUTOMATIC BUSINESS + OWNER PROFILE CREATION -- KC INTEGRATED GLOBAL SERVICES -- ============================================================ -- Remove old trigger/function if they exist drop trigger if exists on_auth_user_created on auth.users; drop function if exists public.handle_new_salespro_user(); -- Create function that runs automatically after signup create or replace function public.handle_new_salespro_user() returns trigger language plpgsql security definer set search_path = public as $$ declare new_business_id uuid; business_name text; full_name text; begin -- Get information supplied during registration business_name := coalesce( new.raw_user_meta_data->>'business_name', 'KC Integrated Global Services' ); full_name := coalesce( new.raw_user_meta_data->>'full_name', new.email ); -- Create the business insert into public.businesses ( name, created_at ) values ( business_name, now() ) returning id into new_business_id; -- Create the owner profile insert into public.profiles ( id, full_name, role, business_id, active, created_at, updated_at ) values ( new.id, full_name, 'owner', new_business_id, true, now(), now() ); return new; exception when others then raise exception 'SalesPro account setup failed: %', SQLERRM; end; $$; -- Automatically execute after every new Auth user create trigger on_auth_user_created after insert on auth.users for each row execute function public.handle_new_salespro_user(); -- ============================================================ -- RLS POLICIES -- ============================================================ alter table public.businesses enable row level security; alter table public.profiles enable row level security; -- Remove possible old policies with these names drop policy if exists "Users can view their business" on public.businesses; drop policy if exists "Users can update their business" on public.businesses; drop policy if exists "Users can view their own profile" on public.profiles; drop policy if exists "Users can update their own profile" on public.profiles; -- BUSINESS: authenticated users can view their own business create policy "Users can view their business" on public.businesses for select to authenticated using ( id in ( select business_id from public.profiles where id = auth.uid() ) ); -- BUSINESS: authenticated users can update their own business create policy "Users can update their business" on public.businesses for update to authenticated using ( id in ( select business_id from public.profiles where id = auth.uid() ) ) with check ( id in ( select business_id from public.profiles where id = auth.uid() ) ); -- PROFILE: user can view own profile create policy "Users can view their own profile" on public.profiles for select to authenticated using ( id = auth.uid() ); -- PROFILE: user can update own profile create policy "Users can update their own profile" on public.profiles for update to authenticated using ( id = auth.uid() ) with check ( id = auth.uid() ); -- ============================================================ -- FINISHED -- ============================================================ select 'SalesPro authentication setup completed successfully.' as message;
